let productionDownloadBusy = false; async function downloadCompleteProductionZip() { if (!productionJobId || productionDownloadBusy) return; const button = document.getElementById( "production-package-btn" ); productionDownloadBusy = true; button.disabled = true; try { await downloadFile({ prefix: "production", url: `${PRODUCTION_API}/download-package/` + encodeURIComponent(productionJobId), filename: `BasinAgentic_Production_Complete_` + `${productionJobId}.zip`, label: "Complete Production ZIP Package", expectedType: "zip" }); } catch (error) { console.error("Production ZIP download:", error); setDownloadProgress( "production", "error", "Production ZIP download failed", 0, 0, error.message ); } finally { productionDownloadBusy = false; button.disabled = false; } } document.getElementById( "production-package-btn" ).addEventListener( "click", downloadCompleteProductionZip );
